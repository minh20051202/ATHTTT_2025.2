  def create_rsa_keypair(self) -> tuple[str, str]:
      """Create an RSA 2048-bit key pair for OAuth2 PKJWT."""
      from cryptography.hazmat.primitives.asymmetric import rsa
      from cryptography.hazmat.primitives import serialization
      from cryptography.hazmat.backends import default_backend
      private_key = rsa.generate_private_key(
          public_exponent=65537,
          key_size=2048,
          backend=default_backend()
      )
      public_key = private_key.public_key()
      private_pem = private_key.private_bytes(
          encoding=serialization.Encoding.PEM,
          format=serialization.PrivateFormat.PKCS8,
          encryption_algorithm=serialization.NoEncryption()
      )
      public_pem = public_key.public_bytes(
          encoding=serialization.Encoding.PEM,
          format=serialization.PublicFormat.SubjectPublicKeyInfo
      )
      return public_pem.decode("utf-8"), private_pem.decode("utf-8")

  // frontend/src/lib/oauth2.js
  /**
   * OAuth2 Private Key JWT helper (client-side).
   * Generates RSA keypair and signs client_assertion JWTs using WebCrypto.
   * No external dependencies — uses SubtleCrypto (available in all modern 
  browsers).
   */

  /**
   * Import a PEM-encoded RSA private key into a CryptoKey for signing.
   * @param {string} privateKeyPem — PEM-encoded PKCS#8 RSA private key
   * @returns {Promise<CryptoKey>}
   */
  export async function importPrivateKey(privateKeyPem) {
    const pemContents = privateKeyPem
      .replace(/-----BEGIN RSA PRIVATE KEY-----/, "")
      .replace(/-----END RSA PRIVATE KEY-----/, "")
      .replace(/-----BEGIN PRIVATE KEY-----/, "")  // PKCS#8 fallback
      .replace(/-----END PRIVATE KEY-----/, "")
      .replace(/\s/g, "");

    const binaryDer = Uint8Array.from(atob(pemContents), (c) =>
  c.charCodeAt(0));

    return crypto.subtle.importKey(
      "pkcs8",
      binaryDer.buffer,
      { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
      false, // not extractable (in production: false; demo can be true)
      ["sign"]
    );
  }

  /**
   * Create a Base64URL-encoded SHA-256 digest of the given data.
   */
  function base64UrlEncode(buffer) {
    return btoa(String.fromCharCode(...new Uint8Array(buffer)))
      .replace(/\+/g, "-")
      .replace(/\//g, "_")
      .replace(/=+$/, "");
  }

  /**
   * Base64URL decode.
   */
  function base64UrlDecode(str) {
    const padded = str + "=".repeat((4 - (str.length % 4)) % 4);
    const base64 = padded.replace(/-/g, "+").replace(/_/g, "/");
    return Uint8Array.from(atob(base64), (c) => c.charCodeAt(0));
  }

  /**
   * Build and sign a JWT assertion (client_assertion) using RS256.
   *
   * @param {string} clientId — iss/sub claim
   * @param {CryptoKey} privateKey — imported RSA private key
   * @param {number} lifetimeMinutes — how long the assertion is valid (default 
  5)
   * @returns {Promise<string>} — signed JWT assertion
   */
  export async function createClientAssertion(clientId, privateKey,
  lifetimeMinutes = 5) {
    const now = Math.floor(Date.now() / 1000);
    const expiry = now + lifetimeMinutes * 60;
    const jti = `${now}-${Math.random().toString(36).slice(2, 10)}`;

    // Header: always {"alg":"RS256","typ":"JWT"}
    const header = { alg: "RS256", typ: "JWT" };
    const headerB64 = base64UrlEncode(new
  TextEncoder().encode(JSON.stringify(header)));

    // Payload
    const payload = {
      iss: clientId,
      sub: clientId,
      aud: "https://agentic-commerce.example/token",
      iat: now,
      exp: expiry,
      jti,
    };
    const payloadB64 = base64UrlEncode(new
  TextEncoder().encode(JSON.stringify(payload)));

    // Signing input: header.payload (base64url, no separator spaces)
    const signingInput = `${headerB64}.${payloadB64}`;

    // Sign with RSASSA-PKCS1-v1_5 + SHA-256
    const signatureBuffer = await crypto.subtle.sign(
      { name: "RSASSA-PKCS1-v1_5" },
      privateKey,
      new TextEncoder().encode(signingInput)
    );
    const signatureB64 = base64UrlEncode(signatureBuffer);

    return `${signingInput}.${signatureB64}`;
  }

  /**
   * Full flow: create client assertion from PEM string.
   * @param {string} clientId
   * @param {string} privateKeyPem — PEM-encoded RSA private key
   * @returns {Promise<string>} — signed JWT assertion
   */
  export async function buildClientAssertion(clientId, privateKeyPem) {
    const privateKey = await importPrivateKey(privateKeyPem);
    return createClientAssertion(clientId, privateKey);
  }
